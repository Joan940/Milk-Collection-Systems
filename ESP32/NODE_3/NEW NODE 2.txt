// Library
#include <SPI.h>
#include <esp_task_wdt.h> // ---> TAMBAHAN 1: Library Watchdog Timer

// Custom Library
#include "LCD.h"
#include "AES.h"
#include "RTC.h"
#include "MESH.h"
#include "RFID.h"
#include "LIDAR.h"
#include "MICROSD.h"
#include "DATABASE.h"

// Konfigurasi WDT
#define WDT_TIMEOUT 30 // ---> TAMBAHAN 2: Batas waktu 30 detik sebelum auto-restart

// Inisialisasi Custom Library
aes myAES;
lcdModule myLcd;
rtcModule myRtc;
rfidModule myRfid;
meshComm myMesh(2);
microSDModule mySD;
lidarModule myLidar;
databaseModule myDb;

// VARIABLE
#define btn1 16
#define btn2 27
#define pinRfidCS 17
#define pinLoraCS 25
#define pinLcdCS 15
#define pinLoraRST 26
#define pinSdCS 13
#define pinPowerRFID 14

bool choice = false;
unsigned long waktuSyncTerakhir = 0;
const float phi = 3.14159265;
const float r = 12.1;

// TAMBAHAN VARIABEL UNTUK STABILITAS SPI
bool layarUtamaTampil = false;
unsigned long waktuCekRFID = 0;

void setup() {
  delay(3000);

  pinMode(pinRfidCS, OUTPUT);
  pinMode(pinLoraCS, OUTPUT);
  pinMode(pinLcdCS, OUTPUT);
  pinMode(pinLoraRST, OUTPUT);
  pinMode(pinSdCS, OUTPUT);

  digitalWrite(pinRfidCS, HIGH);
  digitalWrite(pinLoraCS, HIGH);
  digitalWrite(pinLcdCS, HIGH);
  digitalWrite(pinSdCS, HIGH);
  digitalWrite(pinLoraRST, HIGH);
  delay(100);
  
  Serial.begin(115200);
  Serial.println();

  // PUSH BUTTON
  pinMode(btn1, INPUT_PULLUP);
  pinMode(btn2, INPUT_PULLUP);

  SPI.begin(18, 19, 23);

  // LCD
  myLcd.begin();
  myLcd.centerText("Systems Launch", 1);
  myLcd.show();
  delay(1000);

  // LORA MESH
  myLcd.clear();
  myMesh.begin(false);
  myLcd.centerText("1. Node 2", 1);
  myLcd.show();
  delay(1000);

  // RFID
  myLcd.clear();
  myRfid.begin();
  myLcd.centerText("2. RFID", 1);
  myLcd.show();
  delay(1000);
  
  // MICRO SD
  myLcd.clear();
  mySD.begin();
  myLcd.centerText("3. Micro SD", 1);
  mySD.createNewFile();
  myLcd.show();
  delay(1000);

  // RTC
  myLcd.clear();
  myRtc.begin();
  myLcd.centerText("4. RTC", 1);
  myLcd.show();
  delay(1000);

  // LIDAR
  myLcd.clear();
  myLidar.begin();
  myLcd.centerText("5. LiDAR", 1);
  myLcd.show();
  delay(1000);
  
  // ---> TAMBAHAN 3: Inisialisasi WDT
  esp_task_wdt_init(WDT_TIMEOUT, true); 
  esp_task_wdt_add(NULL); 
}

void loop() {
  esp_task_wdt_reset(); // ---> TAMBAHAN 4: Memberi makan Watchdog Timer

  // ZONA LORA: Bebaskan jalur SPI agar rute Mesh tidak terputus (Cegah Error 2)
  digitalWrite(pinRfidCS, HIGH);
  digitalWrite(pinLcdCS, HIGH);
  digitalWrite(pinSdCS, HIGH);
  myMesh.update();

  if(millis() - waktuSyncTerakhir > 15000) {
    bool adaData = mySD.hasUnsentData();
    String dataTertunda = "";
    if(adaData) {
      dataTertunda = mySD.readNextLine();
    }

    if(adaData && dataTertunda != "") {
      digitalWrite(pinLoraCS, HIGH);
      digitalWrite(pinRfidCS, HIGH);
      myLcd.clear();
      myLcd.centerText("Sinkronisasi...", 1);
      myLcd.show();

      digitalWrite(pinRfidCS, HIGH);
      digitalWrite(pinLcdCS, HIGH);
      digitalWrite(pinSdCS, HIGH);
      delay(50);

      digitalWrite(pinLoraCS, LOW);
      bool statusKirim = myMesh.sendData(dataTertunda);
      digitalWrite(pinLoraCS, HIGH);

      if (statusKirim == true) {
        mySD.removeLine();
        myLcd.clear();
        myLcd.centerText("Sync Berhasil!", 1);
        myLcd.show();
        delay(2000);
      } 

      digitalWrite(pinSdCS, HIGH);
      digitalWrite(pinLoraCS, HIGH);
      delay(50);
      myRfid.rfid.PCD_Init();
      myLcd.begin();
      
      layarUtamaTampil = false; // BUKA KUNCI LCD
    }
    waktuSyncTerakhir = millis(); 
  }

  // ZONA LCD: Digambar 1x saja agar tidak berkedip dan memblokir LoRa
  if (layarUtamaTampil == false) {
    digitalWrite(pinLoraCS, HIGH);
    digitalWrite(pinRfidCS, HIGH);
    digitalWrite(pinSdCS, HIGH);
    
    myLcd.clear();
    myLcd.drawScan();
    myLcd.show();
    
    digitalWrite(pinLcdCS, HIGH);
    layarUtamaTampil = true; 
  }

  // ZONA RFID: Cek setiap 300ms agar LoRa leluasa bernapas
  String currentUID = "";
  if(millis() - waktuCekRFID > 300) {
    digitalWrite(pinLoraCS, HIGH);
    digitalWrite(pinLcdCS, HIGH);
    digitalWrite(pinSdCS, HIGH);
    
    currentUID = myRfid.getUID();
    
    digitalWrite(pinRfidCS, HIGH);
    waktuCekRFID = millis();
  }

  if(currentUID != "") {
    // ---> TAMBAHAN 5: Lepaskan sesi kripto RFID agar bisa tap kartu yang sama berkali-kali
    myRfid.rfid.PICC_HaltA();
    myRfid.rfid.PCD_StopCrypto1();

    layarUtamaTampil = false; // Buka kunci LCD
    
    String username = myDb.getName(currentUID);
    if(username != "Tidak ada") {
      myLcd.clear();
      myLcd.centerText(username, 1);
      myLcd.show();

      delay(1700);

      // KONFIRMASI UID UNTUK LANJUT KE PENGUKURAN
      myLcd.clear();
      myLcd.centerText("MULAI UKUR?", 1);
      myLcd.bottomLeft("TIDAK", 1);
      myLcd.bottomRight("YA", 1);
      myLcd.show();

      float hasilSMA = 0;
      float volumeSusu = 0;
      float volumeMentah = 0;
      while(true) {
        esp_task_wdt_reset(); // ---> TAMBAHAN 6: Cegah reboot saat menunggu input tombol
        myMesh.update();

        if(digitalRead(btn1) == LOW) {
          while(digitalRead(btn1) == LOW) { esp_task_wdt_reset(); delay(100); }
          
          while(true) {
            esp_task_wdt_reset(); // Reset WDT di dalam loop LiDAR

            // LIDAR
            long totalJarak = 0;
            int sampleValid = 0;
            for(int i = 0; i < 30; i++) {
              esp_task_wdt_reset(); // Reset WDT saat proses akuisisi yang panjang
              myMesh.update();

              long jarak = myLidar.jarakMentah();
              if(jarak != -1) {
                totalJarak += jarak;
                sampleValid++;

                myLcd.clear();
                String progress = "";
                for(int j = 0; j < (i % 5) + 1; j++) progress += ".";

                myLcd.centerText(progress, 2);
                myLcd.show();
              }
              delay(100);
            }

            if(sampleValid > 0) {
              hasilSMA = ((float)totalJarak / 30) / 10;
            } else {
              hasilSMA = 0;
            }

            float tinggiSusu = 52.60 - hasilSMA;
            if (tinggiSusu < 0) {
              tinggiSusu = 0;
            }

            volumeSusu = ((phi * pow(r, 2)) * tinggiSusu) / 1000;

            myLcd.clear();
            String hasil = String(volumeSusu);
            myLcd.centerText(hasil, 2);
            myLcd.bottomLeft("ULANG", 1);
            myLcd.bottomRight("LANJUT", 1);
            myLcd.show();

            bool lock = true;
            while(true) {
              esp_task_wdt_reset(); // Reset WDT saat menunggu persetujuan hasil ukur
              myMesh.update();

              if(digitalRead(btn1) == LOW) {
                while(digitalRead(btn1) == LOW) { esp_task_wdt_reset(); delay(100); }
                break;
              }
              if(digitalRead(btn2) == LOW) {
                lock = false;

                myLcd.clear();
                myLcd.show();
                delay(1000);

                while(digitalRead(btn2) == LOW) { esp_task_wdt_reset(); delay(100); }
                break;
              }
              delay(10);
            }

            if(lock == true) {
              break;
            } else {
              continue;
            }
          }
          break;
        }

        if(digitalRead(btn2) == LOW) {
          myLcd.clear();
          myLcd.centerText("Dibatalkan", 1);
          myLcd.show();
          delay(2000);
          layarUtamaTampil = false;
          return;
        }
        delay(10);
      }

      // KONFIRMASI HASIL PENGUKURAN DAN LANJUT PENGIRIMAN
      int lastPressedBtn = 0;

      myLcd.clear();
      myLcd.validation(lastPressedBtn);
      myLcd.show();

      while(true) {
        esp_task_wdt_reset(); // ---> Reset WDT saat proses konfirmasi kirim
        myMesh.update(); 
        
        if(digitalRead(btn1) == LOW) {
          if(lastPressedBtn == 2) {
            String finalData = "2-" + currentUID + "-" + String(volumeSusu) + "-" + myRtc.getTime();
            String dataEnkripsi = myAES.encrypt(finalData);

            unsigned long waktuMulai = millis();
            while(millis() - waktuMulai < 3000) {
              esp_task_wdt_reset();
              myMesh.update();
              myLcd.clear();
              myLcd.drawKirim();
              myLcd.show();
              delay(100);
            }

            myLcd.clear();
            myLcd.show();
            yield();

            // MATIKAN SEMUA SEBELUM LORA BEKERJA
            digitalWrite(pinRfidCS, HIGH);
            digitalWrite(pinLcdCS, HIGH);
            digitalWrite(pinSdCS, HIGH);
            delay(50);
            
            digitalWrite(pinLoraCS, LOW);
            bool statusKirim = myMesh.sendData(dataEnkripsi);
            digitalWrite(pinLoraCS, HIGH);

            // FITUR AUTO-RETRY: Jika gagal (Error 2/5), coba lagi 1x
            if (statusKirim == false) {
              Serial.println("[SYSTEM] Gagal kirim. Mencoba ulang (Retry) dalam 1 detik...");
              delay(1000);
              myMesh.update();
              
              digitalWrite(pinLoraCS, LOW);
              statusKirim = myMesh.sendData(dataEnkripsi);
              digitalWrite(pinLoraCS, HIGH);
            }

            myRfid.rfid.PCD_Init();
            myLcd.begin();

            if (statusKirim == true) {
              myLcd.clear();
              myLcd.pesanTerkirim();
              myLcd.show();
              delay(3000); 
            } else {
              // MATIKAN LORA & RFID SEBELUM SD CARD MENULIS
              digitalWrite(pinLoraCS, HIGH);
              digitalWrite(pinRfidCS, HIGH);
              digitalWrite(pinLcdCS, HIGH);
              delay(50);
              
              mySD.logToFile(dataEnkripsi);
              
              waktuMulai = millis();
              while(millis() - waktuMulai < 3000) {
                esp_task_wdt_reset();
                myLcd.clear();
                myLcd.drawSave();
                myLcd.show();
                delay(10);
              }
            }

            myLcd.clear();
            myLcd.show();
            delay(1000);
            layarUtamaTampil = false;
            break;
          }

          lastPressedBtn = 2;
          myLcd.clear();
          myLcd.validation(lastPressedBtn);
          myLcd.show();
          while(digitalRead(btn1) == LOW) { esp_task_wdt_reset(); delay(100); }
        }

        if(digitalRead(btn2) == LOW) {
          if(lastPressedBtn == 1) {
            myLcd.clear();
            myLcd.centerText("Dibatalkan", 1);
            myLcd.show();
            delay(2000);
            layarUtamaTampil = false;
            break;
          }

          lastPressedBtn = 1;
          myLcd.clear();
          myLcd.validation(lastPressedBtn);
          myLcd.show();
          while(digitalRead(btn2) == LOW) { esp_task_wdt_reset(); delay(500); }
        }

        delay(100);
      }
    } else {
      myLcd.clear();
      myLcd.centerText("AKSES DITOLAK!", 1);
      myLcd.show();
      delay(1700);
      layarUtamaTampil = false;
    }
  }

  delay(10);
}
